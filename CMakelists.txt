cmake_minimum_required(VERSION 3.10)
include(cmake/globals.cmake)

if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(
        CMAKE_TOOLCHAIN_FILE
        "${TOOLCHAIN_ARM_NONE_EABI}"
        CACHE FILEPATH "Toolchain file"
    )
endif()
project(AutoTuner C ASM)

#--------------------------------------#
#           BOARD SELECTION            #
#--------------------------------------#
# Add new boards by following this convention:
# .
# ├── boards/
# │   └── <BOARD>/
# │       └── <BOARD>.cmake   # board-specific build
# │       └── <BOARD>.cfg     # board-specific flash config
# ├── app/                    # application sources
# └── ...                     # other dirs (drivers, system, etc.) 

# Default board
if(NOT DEFINED BOARD)
    set(BOARD "${BOARD_RAPTOR}" CACHE STRING "Target board")
endif()

set(_board_file "${DIR_BOARDS}/${BOARD}/${BOARD}.cmake")

if(EXISTS "${_board_file}")
    include("${_board_file}")
else()
    message(FATAL_ERROR
        "Board '${BOARD}' not found.\n"
        "Expected file: ${_board_file}\n"
        "Each board must live under ${DIR_BOARDS}/<BOARD>/ "
        "and be named <BOARD>.cmake"
    )
endif()

if(NOT DEFINED SRC_BOARD)
    message(FATAL_ERROR
        "Board file did not set SRC_BOARD for BOARD='${BOARD}'.\n"
        "Each board.cmake must assign all its sources to SRC_BOARD."
    )
endif()
if(NOT DEFINED INC_BOARD)
    message(FATAL_ERROR
        "Board file did not set INC_BOARD for BOARD='${BOARD}'.\n"
        "Each <BOARD>.cmake must assign all its include paths to INC_BOARD."
    )
endif()

list(LENGTH SRC_BOARD _sb_len)
if(_sb_len EQUAL 0)
    message(FATAL_ERROR "SRC_BOARD is empty for BOARD='${BOARD}'.")
endif()
foreach(_inc IN LISTS INC_BOARD)
    if(NOT IS_DIRECTORY "${_inc}")
        message(FATAL_ERROR "INC_BOARD contains non-existent dir: ${_inc}")
    endif()
endforeach()

#--------------------------------------#
#            BUILD PROJECT             #
#--------------------------------------#
add_executable(${PROJECT_NAME})


# Aggregate includes (INC_BOARD is a list → no quotes)
target_include_directories(${PROJECT_NAME} PRIVATE
    ${DIR_APP}
    ${DIR_APP_AUTOTUNE}
    ${INC_BOARD}
)

# Aggregate sources (lists → no quotes)
file(GLOB SRC_AUTOTUNE CONFIGURE_DEPENDS "${DIR_APP_AUTOTUNE}/*.c")
target_sources(${PROJECT_NAME} PRIVATE
    ${DIR_APP}/main.c
    ${SRC_AUTOTUNE}
    ${SRC_BOARD}
)

set(FLASH_CFG "${DIR_BOARDS}/${BOARD}/${BOARD}.cfg")
if(NOT EXISTS "${FLASH_CFG}")
  message(FATAL_ERROR "Missing OpenOCD cfg: ${FLASH_CFG}")
endif()

if(DEFINED LIBS_BOARD)
  target_link_libraries(${PROJECT_NAME} PRIVATE ${LIBS_BOARD})
endif()
if(DEFINED DEFS_BOARD)
  target_compile_definitions(${PROJECT_NAME} PRIVATE ${DEFS_BOARD})
endif()
